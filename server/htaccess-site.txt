# .htaccess da pasta do jogo na hospedagem (vai para a mesma pasta do index.html)
AddType application/wasm .wasm
AddType application/octet-stream .pck
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE application/wasm application/javascript text/html application/json application/octet-stream
</IfModule>
# config.php só é lido pelo api.php, nunca pelo navegador
<Files "config.php">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order deny,allow
    Deny from all
  </IfModule>
</Files>
# arquivos .ht* (este aqui) também não saem
<FilesMatch "^\.ht">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order deny,allow
    Deny from all
  </IfModule>
</FilesMatch>
